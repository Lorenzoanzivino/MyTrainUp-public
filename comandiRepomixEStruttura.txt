# 1. Aggiorna il codice Frontend
repomix frontend --output repomix-frontend.md --ignore "**/node_modules/**,**/dist/**,**/.git/**,**/package-lock.json"

# 2. Aggiorna il codice Backend (senza database e dati sensibili)
repomix backend --output repomix-backend.md --ignore "**/venv/**,**/__pycache__/**,**/*.db,**/.env,**/static/uploads/**,**/.git/**"

# 3. Aggiorna la struttura (Profonda, SENZA limiti di livello)
tree -I "node_modules|venv|__pycache__|dist|.git|package-lock.json" > struttura-progetto.txt

git add repomix-frontend.md repomix-backend.md struttura-progetto.txt
git commit -m "docs: aggiornamento documentazione repomix e struttura profonda"
git push origin main
